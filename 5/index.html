def iterative_denoise(im_noisy, i_start, prompt_embeds, timesteps, display=True):
  image = im_noisy
  finalfinal = None
  with torch.no_grad():
    for i in range(i_start, len(timesteps) - 1):
      # Get timesteps
      t = timesteps[i]
      prev_t = timesteps[i+1]

      # get `alpha_cumprod` and `alpha_cumprod_prev` for timestep t from `alphas_cumprod`
      # compute `alpha`
      # compute `beta`
      # ===== your code here! =====
      a = alphas_cumprod[t]/alphas_cumprod[prev_t]
      b = (1 - a)
      # ==== end of code ====

      # Get noise estimate
      model_output = stage_1.unet(
          image,
          t,
          encoder_hidden_states=prompt_embeds,
          return_dict=False
      )[0]

      # Split estimate into noise and variance estimate
      noise_est, predicted_variance = torch.split(model_output, image.shape[1], dim=1)

      # compute `pred_prev_image` (x_{t'}), the DDPM estimate for the image at the
      # next timestep, which is slightly less noisy. Use the equation 3.
      # This is the core of DDPM
      # ===== your code here! =====
      x0 = (image - torch.sqrt(1-alphas_cumprod[t].to(image.device)) * noise_est) / torch.sqrt(alphas_cumprod[t].to(image.device))
      a1 = alphas_cumprod[t].to(image.device)
      a2 = alphas_cumprod[prev_t].to(image.device)
      img = torch.sqrt(a2) * b / (1 - a1) * x0 + torch.sqrt(a) * (1 - a2) / (1 - a1) * image
      pred_prev_image = add_variance(predicted_variance, t, img)
      # ==== end of code ====

      image = pred_prev_image
    clean = image.cpu().detach().numpy()

  return clean

1.6: CFG

Next, I implemented CFG according to the algorithm described in the spec. I did this by running the UNet twice, and then combining the noise amounts. I then added conditional variance to the output, and resulting images were much better quality then the diffusion model sampling images. My code for this section is below

def iterative_denoise_cfg(im_noisy, i_start, prompt_embeds, uncond_prompt_embeds, timesteps, scale=7, display=True):
  image = im_noisy

  with torch.no_grad():
    for i in range(i_start, len(timesteps) - 1):
      # Get timesteps
      t = timesteps[i]
      prev_t = timesteps[i+1]

      # Get `alpha_cumprod`, `alpha_cumprod_prev`, `alpha`, `beta`
      # ===== your code here! =====
      a1 = alphas_cumprod[t].to(image.device)
      a2 = alphas_cumprod[prev_t].to(image.device)
      b = (1 - a1/a2)

      # ==== end of code ====

      # Get cond noise estimate
      model_output = stage_1.unet(
          image,
          t,
          encoder_hidden_states=prompt_embeds,
          return_dict=False
      )[0]

      # Get uncond noise estimate
      uncond_model_output = stage_1.unet(
          image,
          t,
          encoder_hidden_states=uncond_prompt_embeds,
          return_dict=False
      )[0]

      # Split estimate into noise and variance estimate
      noise_est, predicted_variance = torch.split(model_output, image.shape[1], dim=1)
      uncond_noise_est, _ = torch.split(uncond_model_output, image.shape[1], dim=1)

      # Compute the CFG noise estimate based on equation 4
      # ===== your code here! =====
      cfg = uncond_noise_est + scale * (noise_est - uncond_noise_est)
      # ==== end of code ====


      # Get `pred_prev_image`, the next less noisy image.
      # ===== your code here! =====
      x0 = (image - torch.sqrt(1-a1) * cfg) / torch.sqrt(a1)

      pred_prev_image = add_variance(predicted_variance, t, (torch.sqrt(a2) * b / (1-a1) * x0) + torch.sqrt(a1/a2) * (1-a2)/(1-a1) * image)
      # ==== end of code ====

      image = pred_prev_image
    clean = image.cpu().detach().numpy()

  return clean

1.7.2: Inpainting

I used the code below to generate these images. The main idea of the code is to start from a fully noisy image and for each timestep, blend the 2 noise predictions and then reinsert the original image outside the mask by re-noising it to the current timestep and combining it with the generated area. 

          
def inpaint(original_image, mask, prompt_embeds, uncond_prompt_embeds, timesteps, scale=7, display=True):
  image = torch.randn_like(original_image).to(device).half()
  with torch.no_grad():
    for i in range(len(timesteps) - 1):
      t = timesteps[i]
      prev_t = timesteps[i+1]
      a1 = alphas_cumprod[t].to(image.device)
      a2 = alphas_cumprod[prev_t].to(image.device)
      b = (1 - a1/a2)
      model_output = stage_1.unet(
          image,
          t,
          encoder_hidden_states=prompt_embeds,
          return_dict=False
      )[0]

      uncond_model_output = stage_1.unet(image, t, encoder_hidden_states=uncond_prompt_embeds, return_dict=False)[0]
      noise_est, predicted_variance = torch.split(model_output, image.shape[1], dim=1)
      uncond_noise_est, _ = torch.split(uncond_model_output, image.shape[1], dim=1)
      cfg = uncond_noise_est + scale * (noise_est - uncond_noise_est)

      x0 = (image - torch.sqrt(1-a1) * cfg) / torch.sqrt(a1)
      pred_prev_image = add_variance(predicted_variance, t, (torch.sqrt(a2) * b / (1-a1) * x0) + torch.sqrt(a1/a2) * (1-a2)/(1-a1) * image)



      noisy = forward(original_image, prev_t).half().to(device)
      image = pred_prev_image* mask + noisy*(1-mask)

1.8: Visual Algorithms

To make a visual anagram, I repeated a similar process above, but each iteration, I flipped the image once and generated an image for a given prompt for the right-side up image, and also generated an image for a different prompt for the upside-down version of the image. Below is the code:

          
def make_flip_illusion(image, i_start, prompt_embeds, uncond_prompt_embeds, timesteps, scale=7, display=True):
  e1, e2 = prompt_embeds[0], prompt_embeds[1]
  with torch.no_grad():
    t_start = timesteps[i_start]
    image = forward(image, t_start).half().to(device)
    for i in range(i_start, len(timesteps) - 1):
      t = timesteps[i]
      prev_t = timesteps[i+1]
      a1 = alphas_cumprod[t].to(image.device)
      a2 = alphas_cumprod[prev_t].to(image.device)
      b = (1 - a1/a2)
      model_output = stage_1.unet(
          image,
          t,
          encoder_hidden_states=e1,
          return_dict=False
      )[0]
      uncond_model_output = stage_1.unet(image, t, encoder_hidden_states=uncond_prompt_embeds, return_dict=False)[0]
      flipped = torch.flip(image, dims=(2,))
      model_output_flipped = stage_1.unet(flipped, t, encoder_hidden_states=e2, return_dict=False)[0]
      uncond_model_output_flipped = stage_1.unet(flipped, t, encoder_hidden_states=uncond_prompt_embeds, return_dict=False)[0]

      noise_est, predicted_variance = torch.split(model_output, image.shape[1], dim=1)
      uncond_noise_est, _ = torch.split(uncond_model_output, image.shape[1], dim=1)
      cfg = uncond_noise_est + scale * (noise_est - uncond_noise_est)

      noise_est_flipped, _ = torch.split(model_output_flipped, image.shape[1], dim=1)
      uncond_noise_est_flipped, _ = torch.split(uncond_model_output_flipped, image.shape[1], dim=1)
      cfg_flipped = uncond_noise_est_flipped + scale * (noise_est_flipped - uncond_noise_est_flipped)
      cfg_flipped = torch.flip(cfg_flipped, dims=(2,))

      cfg = (cfg + cfg_flipped) / 2
      x0 = (image - torch.sqrt(1-a1) * cfg) / torch.sqrt(a1)

      
      pred_prev_image = add_variance(predicted_variance, t, (torch.sqrt(a2) * b / (1-a1) * x0) + torch.sqrt(a1/a2) * (1-a2)/(1-a1) * image)

    
      image = pred_prev_image

    return image

1.9: Hybrid Images

To make hybrid images, I combined the low freq noise which was predicted from the first prompt with the high freq from the other prompt at each iteration step. Then, I used that combined noise to guide the denoising process, as done before.

      a1 = alphas_cumprod[t].to(image.device)
      a2 = alphas_cumprod[prev_t].to(image.device)
      b = (1 - a1/a2)
      model_output1 = stage_1.unet(

      n1, var = torch.split(model_output1, image.shape[1], dim=1)
      uncondn1, _ = torch.split(uncond_model_output1, image.shape[1], dim=1)
      cfg1 = uncondn1 + scale * (n1 - uncondn1)

          image,
